using Hangfire;
using Microsoft.EntityFrameworkCore;
using Microsoft.Extensions.Logging;
using Serilog.Context;
using TaxReader.Application.Interfaces;
using TaxReader.Domain.Entities;
using TaxReader.Domain.Enums;

namespace TaxReader.Application.Jobs;

public class GrantTokensJob(IAppDbContext dbContext, ILogger&lt;GrantTokensJob&gt; logger)
{
    [AutomaticRetry(Attempts = 3, DelaysInSeconds = new[] { 30, 120, 300 })]
    public async Task HandleAsync(Guid userId, int credits, CancellationToken cancellationToken)
    {
        using var _scope = LogContext.PushProperty("JobId", $"Grant_{userId}_{credits}");

        var balance = await dbContext.UserTokenBalances
            .FirstOrDefaultAsync(b =&gt; b.UserId == userId, cancellationToken);

        if (balance is null)
        {
            balance = new UserTokenBalance
            {
                Id = Guid.NewGuid(),
                UserId = userId,
                UserKey = userId.ToString(),
                Balance = 0,
                UpdatedAt = DateTime.UtcNow
            };
            dbContext.UserTokenBalances.Add(balance);
        }

        balance.Balance += credits;
        balance.UpdatedAt = DateTime.UtcNow;

        dbContext.TokenTransactions.Add(new TokenTransaction
        {
            Id = Guid.NewGuid(),
            UserId = userId,
            UserKey = userId.ToString(),
            Type = TokenTransactionType.Purchase,
            Amount = credits,
            BalanceAfter = balance.Balance,
            Description = $"Zahlung bestätigt: {credits} Credits",
            CreatedAt = DateTime.UtcNow
        });

        // Update the most recent Pending payment for this user to Granted
        var payment = await dbContext.Payments
            .Where(p =&gt; p.UserId == userId && p.Status == PaymentStatus.Pending && p.CreditsGranted == credits)
            .OrderByDescending(p =&gt; p.CreatedAt)
            .FirstOrDefaultAsync(cancellationToken);
        if (payment is not null)
        {
            payment.Status = PaymentStatus.Granted;
        }

        await dbContext.SaveChangesAsync(cancellationToken);
        logger.LogInformation("Granted {Credits} tokens to User {UserId} — new balance: {Balance}",
            credits, userId, balance.Balance);
    }

    public static RouteGroupBuilder MapPaymentEndpoints(this RouteGroupBuilder group)
    {
        var payments = group.MapGroup("/payments").WithTags("Payments");

        payments.MapPost("/checkout", async (
            CreateCheckoutSessionRequest request,
            ICurrentUser currentUser,
            IStripePaymentProvider stripeProvider,
            ITokenService tokenService,
            IAppDbContext dbContext,
            IOptions&lt;StripeOptions&gt; stripeOptions,
            CancellationToken cancellationToken) =&gt;
        {
            if (!request.WaiverAccepted || !request.AgbAccepted)
                return Results.BadRequest(new { error = "Bitte akzeptieren Sie die AGB und den Widerrufsverzicht." });

            var validCredits = new[] { 50, 200, 500 };
            if (!validCredits.Contains(request.Credits))
                return Results.BadRequest(new { error = "Ungültige Credits-Anzahl." });

            var opts = stripeOptions.Value;

            // D-14: DemoMode — skip Stripe, credit directly
            if (opts.DemoMode)
            {
                await tokenService.AddTokensAsync(
                    request.Credits,
                    TokenTransactionType.Purchase,
                    $"Demo-Kauf: {request.Credits} Credits",
                    cancellationToken);
                return Results.Ok(new CheckoutSessionDto(

                    .FirstOrDefaultAsync(u =&gt; u.Id == currentUser.UserId, cancellationToken);

                return Results.Ok(new CheckoutSessionDto(session));
            }

            {
                return Results.BadRequest(new { error = ex.Message });
            }
        })
        .WithName("CreateCheckoutSession")
        .WithSummary("Create a Stripe Checkout session for a token pack purchase");

        payments.MapGet("/invoices", async (
            ICurrentUser currentUser,

    /// &lt;summary&gt;
    /// D-15: Webhook endpoint — anonymous, NOT under /api/v1 auth group.
    /// Raw body must be read before any JSON model binding consumes the stream.
    /// Accepts HttpRequest directly so ASP.NET does not consume the body (Pitfall 1).
    /// &lt;/summary&gt;
    public static WebApplication MapStripeWebhookEndpoint(this WebApplication app)
    {
        app.MapPost("/webhooks/stripe", async (
            HttpRequest request,
            IOptions&lt;StripeOptions&gt; stripeOptions,
            IAppDbContext dbContext,
            IBackgroundJobClient jobClient,
            ILogger&lt;PaymentEndpoints&gt; logger,
            CancellationToken cancellationToken) =&gt;
        {
            var json = await new System.IO.StreamReader(request.Body).ReadToEndAsync(cancellationToken);
            var sig = request.Headers["Stripe-Signature"].ToString();

            Stripe.Event stripeEvent;
            try
            {
                stripeEvent = Stripe.EventUtility.ConstructEvent(
                    json, sig, stripeOptions.Value.WebhookSecret);
            }
            catch (Stripe.StripeException ex)
            {
                logger.LogWarning("Stripe webhook signature validation failed: {Message}", ex.Message);
                return Results.BadRequest();
            }

            if (stripeEvent.Type == Stripe.EventTypes.CheckoutSessionCompleted)
            {
                var session = stripeEvent.Data.Object as Stripe.Checkout.Session;
                if (session is null) return Results.Ok();

                var stripeEventId = stripeEvent.Id;
                // T-02: idempotency — UNIQUE constraint on stripe_event_id
                var alreadyProcessed = await dbContext.Payments
                    .AnyAsync(p =&gt; p.StripeEventId == stripeEventId, cancellationToken);
                if (alreadyProcessed)
                {
                    logger.LogInformation("Duplicate Stripe event {StripeEventId} — ignoring", stripeEventId);
                    return Results.Ok(); // Pitfall 2: always 200 on duplicate
                }

                if (!session.Metadata.TryGetValue("userId", out var userIdStr) ||
                    !Guid.TryParse(userIdStr, out var userId))
                {
                    logger.LogWarning("Stripe event {StripeEventId} missing userId metadata", stripeEventId);
                    return Results.Ok();
                }

                if (!session.Metadata.TryGetValue("credits", out var creditsStr) ||
                    !int.TryParse(creditsStr, out var credits))
                {
                    logger.LogWarning("Stripe event {StripeEventId} missing credits metadata", stripeEventId);
                    return Results.Ok();
                }

                // Persist stripe_customer_id on User for future portal/invoice calls (Pitfall 5)
                if (session.CustomerId is not null)
                {
                    var user = await dbContext.Users
                        .FirstOrDefaultAsync(u =&gt; u.Id == userId, cancellationToken);
                    if (user is not null && user.StripeCustomerId is null)
                    {
                        user.StripeCustomerId = session.CustomerId;
                    }
                }

                dbContext.Payments.Add(new Payment
                {
                    Id = Guid.NewGuid(),
                    UserId = userId,
                    StripeEventId = stripeEventId,
                    StripeSessionId = session.Id,
                    // Populate StripePaymentIntentId for reliable charge.refunded correlation in Plan 05-04
                    StripePaymentIntentId = session.PaymentIntentId,
                    CreditsGranted = credits,
                    AmountCents = (int)(session.AmountTotal ?? 0),
                    Currency = session.Currency ?? "eur",
                    Status = PaymentStatus.Pending,
                    CreatedAt = DateTime.UtcNow
                });
                await dbContext.SaveChangesAsync(cancellationToken);

                await jobClient.EnqueueAsync&lt;GrantTokensJob&gt;(
                    j =&gt; j.HandleAsync(userId, credits, CancellationToken.None),
                    cancellationToken);

                logger.LogInformation(
                    "Processed checkout.session.completed for User {UserId}, {Credits} credits, event {StripeEventId}",
                    userId, credits, stripeEventId);
            }
            else if (stripeEvent.Type == Stripe.EventTypes.ChargeRefunded)
            {
                // RevokeTokensJob wired in Plan 05-04
                logger.LogInformation("Received charge.refunded event {StripeEventId} — RevokeTokensJob not yet registered", stripeEvent.Id);
            }

            return Results.Ok();
        })
        .AllowAnonymous()
        .WithTags("Webhooks");

        return app;
    }
}
